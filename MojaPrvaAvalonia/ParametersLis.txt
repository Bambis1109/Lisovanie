    private double smernicaK;
    [ReadOnly(true), DisplayName("Smernik"), Description("Smernik linearnej interpolacie.")]
    public double SmernicaK { get => smernicaK; set => SetProperty(ref smernicaK, value, nameof(SmernicaK)); }
    private double konstantaB;
    [ReadOnly(true), DisplayName("Offset"), Description("Posun linearnej interpolacie.")]
    public double KonstantaB { get => konstantaB; set => SetProperty(ref konstantaB, value, nameof(KonstantaB)); }

    // VÝPOČET KOEFICIENTOV Z DVOCH KALIBRAČNÝCH BODOV
    public void RecalculateCalibration()
    {
        // y = VyskaKalibra - VyskaSenAbsolut (predĺženie; kladné pri klesajúcej vzdialenosti senzora so silou)
        double y1 = VyskaKalibra - VyskaSenAbsolut1;
        double y2 = VyskaKalibra - VyskaSenAbsolut2;

        double dx = SilaKalib2 - SilaKalib1;
        if (Math.Abs(dx) < 1e-12)
            throw new ArgumentException("SilaKalib1 a SilaKalib2 musia byť rôzne.");

        // Smernica a konštanta priamky y = K*F + B
        double K = (y2 - y1) / dx;
        double B = y1 - K * SilaKalib1;

        SmernicaK = K;
        KonstantaB = B;
    }

    // PREDPOVEDANÉ PREDĽŽENIE PRE ĽUBOVOĽNÚ SILU (voliteľná pomocná metóda)
    public double PredictElongation(double sila)
    {
        return SmernicaK * sila + KonstantaB; // y(F)
    }

    // PREPOČÍTANÁ VZDIALENOSŤ (odľahčená/normalizovaná na „bez sily“)
    public double RecomputedDistance(double sila, double senAbsolut)
    {
        // senAbsolut ... aktuálna absolútna vzdialenosť senzora pri sile 'sila'
        // y = predĺženie; meraná vzdialenosť klesá o y, preto bezzaťažová ~ senAbsolut + y
        double y = PredictElongation(sila);
        double s0 = senAbsolut + y;
        return Math.Round(s0, 2); // zaokrúhlenie na 2 desatinné miesta
    }
